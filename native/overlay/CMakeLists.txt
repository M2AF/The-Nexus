cmake_minimum_required(VERSION 3.21)
project(NexusOverlay LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(NexusOverlay SHARED bridge.cpp renderer.cpp)
target_compile_definitions(NexusOverlay PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(NexusOverlay PRIVATE minhook d3d11 d3d12 dxgi d3dcompiler advapi32 user32)
add_executable(NexusOverlayTest test-game.cpp)
target_compile_definitions(NexusOverlayTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(NexusOverlayTest PRIVATE d3d12 dxgi user32)
